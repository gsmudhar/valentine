  <script>

    // ====== Customize these ======

    const HER_NAME = "Babe";

    const YOUR_NAME = "Sunny";

    const WIN_MESSAGE = `Okay ${HER_NAME}, it’s official. 💗

Dinner is on me. You pick dessert.

Love, ${YOUR_NAME}`;

    // ============================


    const stage = document.getElementById("stage");

    const noBtn = document.getElementById("no");

    const yesBtn = document.getElementById("yes");

    const resetBtn = document.getElementById("reset");

    const success = document.getElementById("success");

    const msg = document.getElementById("message");

    const subtitle = document.getElementById("subtitle");


    msg.textContent = WIN_MESSAGE;


    // Make sure "No" starts somewhere valid

    function placeNoButton(x, y){

      const s = stage.getBoundingClientRect();

      const b = noBtn.getBoundingClientRect();


      // clamp so it stays inside stage

      const pad = 10;

      const maxX = s.width - b.width - pad;

      const maxY = s.height - b.height - pad;


      const nx = Math.min(Math.max(x, pad), maxX);

      const ny = Math.min(Math.max(y, pad), maxY);


      noBtn.style.left = nx + "px";

      noBtn.style.top  = ny + "px";

      noBtn.style.transform = "translate(0,0)";

    }


    function randomSpotFarFrom(cursorX, cursorY){

      const s = stage.getBoundingClientRect();

      const b = noBtn.getBoundingClientRect();

      const pad = 10;


      const spots = [];

      for(let i=0;i<24;i++){

        const x = pad + Math.random() * (s.width - b.width - pad*2);

        const y = pad + Math.random() * (s.height - b.height - pad*2);

        const dx = x - cursorX, dy = y - cursorY;

        const dist = Math.hypot(dx, dy);

        spots.push({x,y,dist});

      }

      spots.sort((a,b)=>b.dist - a.dist);

      return spots[0];

    }


    // "No" dodges when cursor gets near

    function maybeDodge(clientX, clientY){

      const s = stage.getBoundingClientRect();

      const b = noBtn.getBoundingClientRect();


      // cursor position relative to stage

      const cx = clientX - s.left;

      const cy = clientY - s.top;


      // button center relative to stage

      const bx = (b.left - s.left) + b.width/2;

      const by = (b.top  - s.top)  + b.height/2;


      const dist = Math.hypot(cx - bx, cy - by);


      // tweak this: smaller = harder to click No

      const DODGE_RADIUS = 90;


      if(dist < DODGE_RADIUS){

        const spot = randomSpotFarFrom(cx, cy);

        placeNoButton(spot.x, spot.y);

        subtitle.textContent = "Hmm… interesting choice 😌";

      }

    }


    // Desktop: move on mouse proximity

    stage.addEventListener("mousemove", (e) => maybeDodge(e.clientX, e.clientY));


    // Mobile: dodge when they try to tap it (touchstart)

    noBtn.addEventListener("touchstart", (e) => {

      const t = e.touches[0];

      maybeDodge(t.clientX, t.clientY);

      e.preventDefault();

    }, {passive:false});


    // If they somehow click No

    noBtn.addEventListener("click", () => {

      subtitle.textContent = "Nice try 😂";

      // immediately move it anyway

      const s = stage.getBoundingClientRect();

      placeNoButton(20 + Math.random()*(s.width-120), 20 + Math.random()*(s.height-80));

    });


    // Yes flow

    yesBtn.addEventListener("click", () => {

      success.style.display = "block";

      subtitle.textContent = "Correct answer 😇";

      burst();

    });


    resetBtn.addEventListener("click", () => {

      success.style.display = "none";

      subtitle.textContent = "Choose wisely 🙂";

      // re-center-ish

      placeNoButton(stage.clientWidth*0.55, stage.clientHeight*0.55);

      clearConfetti();

    });


    // initial placement after layout

    requestAnimationFrame(() => {

      placeNoButton(stage.clientWidth*0.55, stage.clientHeight*0.55);

    });


    // ---- confetti ----

    const canvas = document.getElementById("fx");

    const ctx = canvas.getContext("2d");

    const card = document.getElementById("card");

    let W=0,H=0, conf=[];

    function resize(){

      const r = card.getBoundingClientRect();

      canvas.width = Math.floor(r.width * devicePixelRatio);

      canvas.height = Math.floor(r.height * devicePixelRatio);

      canvas.style.width = r.width + "px";

      canvas.style.height = r.height + "px";

      W = canvas.width; H = canvas.height;

      ctx.setTransform(1,0,0,1,0,0);

    }

    window.addEventListener("resize", resize);

    resize();


    function burst(){

      conf = [];

      const colors = ["#ff4d6d","#ffd166","#8ecae6","#cdb4db","#a7f3d0","#ffffff"];

      for(let i=0;i<160;i++){

        conf.push({

          x: W/2 + (Math.random()-0.5)*80,

          y: H*0.22 + (Math.random()-0.5)*40,

          vx: (Math.random()-0.5)*11,

          vy: Math.random()*-10 - 4,

          g: 0.26 + Math.random()*0.12,

          s: 3 + Math.random()*4.5,

          a: 1,

          c: colors[(Math.random()*colors.length)|0],

          r: Math.random()*Math.PI

        });

      }

    }

    function clearConfetti(){ conf = []; }


    function tick(){

      ctx.clearRect(0,0,W,H);

      for(const p of conf){

        p.vy += p.g;

        p.x += p.vx;

        p.y += p.vy;

        p.r += 0.12;

        p.a *= 0.993;

        ctx.globalAlpha = p.a;

        ctx.fillStyle = p.c;

        ctx.save();

        ctx.translate(p.x,p.y);

        ctx.rotate(p.r);

        ctx.fillRect(-p.s/2, -p.s/2, p.s, p.s*1.6);

        ctx.restore();

      }

      ctx.globalAlpha = 1;

      requestAnimationFrame(tick);

    }

    tick();
